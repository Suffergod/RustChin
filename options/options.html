pub fn calculate_direction(text: &str) -> &'static str {
    let rtl_chars = text.chars().filter(|c| is_persian_or_arabic(*c)).count();
    let ltr_chars = text.chars().filter(|c| c.is_ascii_alphabetic()).count();
    if rtl_chars >= ltr_chars { "rtl" } else { "ltr" }
}
